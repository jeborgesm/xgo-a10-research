cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(xgo_happy C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(xgo_happy_a main.c)
target_compile_definitions(xgo_happy_a PRIVATE CLOCK_IS_DP=1)
target_link_libraries(xgo_happy_a pico_stdlib)
pico_enable_stdio_usb(xgo_happy_a 0)
pico_enable_stdio_uart(xgo_happy_a 0)
pico_add_extra_outputs(xgo_happy_a)
add_executable(xgo_happy_b main.c)
target_compile_definitions(xgo_happy_b PRIVATE CLOCK_IS_DP=0)
target_link_libraries(xgo_happy_b pico_stdlib)
pico_enable_stdio_usb(xgo_happy_b 0)
pico_enable_stdio_uart(xgo_happy_b 0)
pico_add_extra_outputs(xgo_happy_b)
