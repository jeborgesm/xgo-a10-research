cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(xgo_live_controller C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(xgo_live_controller main.c)
target_link_libraries(xgo_live_controller pico_stdlib hardware_resets)
pico_enable_stdio_usb(xgo_live_controller 0)
pico_enable_stdio_uart(xgo_live_controller 0)
pico_add_extra_outputs(xgo_live_controller)
