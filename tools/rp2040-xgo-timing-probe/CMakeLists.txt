cmake_minimum_required(VERSION 3.13)

project(xgo_timing_probe C CXX ASM)

pico_enable_stdio_usb(xgo_timing_probe 0)
pico_enable_stdio_uart(xgo_timing_probe 0)
pico_add_extra_outputs(xgo_timing_probe)
