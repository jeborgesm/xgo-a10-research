cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(xgo_p0 C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(xgo_p0 main.c)
pico_generate_pio_header(xgo_p0 ${CMAKE_CURRENT_LIST_DIR}/xgo_sampler.pio)
target_link_libraries(xgo_p0 pico_stdlib hardware_pio)
pico_enable_stdio_usb(xgo_p0 1)
pico_enable_stdio_uart(xgo_p0 0)
pico_add_extra_outputs(xgo_p0)
